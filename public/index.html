async function sendGroupWhatsAppDirect() { const day = days.find(item => item.id === currentDayId); const element = document.getElementById('capture-area'); if (!day || !element) { alert('Der Tagesplan konnte nicht gefunden werden.'); return; } if (typeof html2canvas === 'undefined') { alert('Die Screenshot-Bibliothek ist noch nicht geladen. Bitte kurz warten und erneut versuchen.'); return; } try { const canvas = await html2canvas(element, { scale: 2, backgroundColor: '#ffffff', useCORS: true }); const blob = await new Promise(resolve => { canvas.toBlob(resolve, 'image/png'); }); if (!blob) { throw new Error('PNG konnte nicht erzeugt werden.'); } // Bild direkt in die Zwischenablage kopieren (ohne Download) let copiedToClipboard = false; if ( navigator.clipboard && typeof ClipboardItem !== 'undefined' && window.isSecureContext ) { try { const clipboardItem = new ClipboardItem({ 'image/png': blob }); await navigator.clipboard.write([clipboardItem]); copiedToClipboard = true; } catch (clipboardError) { console.warn( 'Kopieren in die Zwischenablage fehlgeschlagen:', clipboardError ); } } // WhatsApp im neuen Tab öffnen const text = `*${day.name}, ${day.date}*`; const whatsappUrl = `https://api.whatsapp.com/send?text=${encodeURIComponent(text)}`; window.open(whatsappUrl, '_blank', 'noopener,noreferrer'); if (copiedToClipboard) { alert( 'Der Dienstplan wurde in die Zwischenablage kopiert!\n\n' + 'Öffne den WhatsApp-Chat und drücke einfach STRG + V (oder halte das Eingabefeld gedrückt), um das Bild einzufügen.' ); } else { alert( 'Dein Browser blockiert das direkte Kopieren in die Zwischenablage. Bitte erlaube den Zugriff in den Browsereinstellungen.' ); } } catch (error) { console.error('Fehler beim Erstellen des Dienstplans:', error); alert( 'Der Dienstplan konnte nicht erstellt werden.\n\n' + 'Details: ' + (error.message || error) ); } }
